Sync Document Payload همگام‌سازی و بارگذاری داده‌های سند

Transmits serialized entity data to the processing server for cryptographic validation and storage. ارسال ساختار داده‌های سریا‌ل‌شده به سرور پردازشی جهت اعتبارسنجی رمزنگاری و ثبت در پایگاه داده.

03 · Code Implementationپیاده‌سازی کد فنی
JS sync_client.js 
// Code block is ALWAYS LTR regardless of page direction
export async function syncDocument(endpoint, docId, token) {
  const response = await fetch(endpoint, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify({ document_id: docId, compress_mode: true })
  });
  return await response.json();
}
